cmake_minimum_required(VERSION 3.21)
project(pando_qgis_bridge LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT PANDO_CONSUMER_ROOT)
  message(FATAL_ERROR "Set PANDO_CONSUMER_ROOT to the pinned Pando computation dependency")
endif()
file(TO_CMAKE_PATH "${PANDO_CONSUMER_ROOT}" PANDO_CONSUMER_ROOT)
if(TRUE)
    if(NOT WIN32 OR NOT PANDO_QGIS_SDK_ROOT)
        message(FATAL_ERROR "The QGIS bridge needs the matching Windows OSGeo4W SDK root")
    endif()
    set(PANDO_QGIS_DIR "${PANDO_QGIS_SDK_ROOT}/apps/qgis-ltr")
    set(PANDO_QT_DIR "${PANDO_QGIS_SDK_ROOT}/apps/qt5")
    file(GLOB PANDO_QT_ENTRIES LIST_DIRECTORIES true "${PANDO_QT_DIR}/include/*")
    set(PANDO_QT_INCLUDE_DIRS)
    foreach(entry IN LISTS PANDO_QT_ENTRIES)
        if(IS_DIRECTORY "${entry}")
            list(APPEND PANDO_QT_INCLUDE_DIRS "${entry}")
        endif()
    endforeach()
    add_library(pando-qgis-bridge MODULE pando-qgis-plugin.cpp qgis-export.cpp "${PANDO_CONSUMER_ROOT}/src/data/native/src/losslessjson.cpp")
    set_target_properties(pando-qgis-bridge PROPERTIES PREFIX "" OUTPUT_NAME "pando_qgis_bridge")
    target_include_directories(pando-qgis-bridge PRIVATE ${PANDO_CONSUMER_ROOT}/src/document/native/include ${PANDO_CONSUMER_ROOT}/src/geometry/native/include ${PANDO_CONSUMER_ROOT}/src/map/native/include ${PANDO_CONSUMER_ROOT}/src/rendering/native/include ${PANDO_CONSUMER_ROOT}/src/data/native/include ${PANDO_CONSUMER_ROOT}/src/runtime/native/include "${PANDO_QGIS_DIR}/include" "${PANDO_QGIS_SDK_ROOT}/include"
        "${PANDO_QT_DIR}/include" ${PANDO_QT_INCLUDE_DIRS})
    target_link_libraries(pando-qgis-bridge PRIVATE "${PANDO_QGIS_DIR}/lib/qgis_core.lib" "${PANDO_QGIS_DIR}/lib/qgis_gui.lib"
        "${PANDO_QT_DIR}/lib/Qt5Core.lib" "${PANDO_QT_DIR}/lib/Qt5Network.lib" "${PANDO_QT_DIR}/lib/Qt5Widgets.lib" "${PANDO_QT_DIR}/lib/Qt5Gui.lib")
    target_compile_definitions(pando-qgis-bridge PRIVATE QT_CORE_LIB QT_NETWORK_LIB QT_WIDGETS_LIB QT_GUI_LIB _USE_MATH_DEFINES)
    if(MSVC)
        target_compile_options(pando-qgis-bridge PRIVATE /utf-8)
    endif()
endif()
