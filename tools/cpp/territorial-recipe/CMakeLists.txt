cmake_minimum_required(VERSION 3.21)
project(pando_historical_recipe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT PANDO_CONSUMER_ROOT)
  message(FATAL_ERROR "Set PANDO_CONSUMER_ROOT to the pinned Pando computation dependency")
endif()
file(TO_CMAKE_PATH "${PANDO_CONSUMER_ROOT}" PANDO_CONSUMER_ROOT)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GEOS REQUIRED IMPORTED_TARGET geos)
pkg_check_modules(PROJ REQUIRED IMPORTED_TARGET proj)
add_executable(pando-territorial-recipe build-territorial-recipe.cpp "${PANDO_CONSUMER_ROOT}/src/data/native/src/losslessjson.cpp" "${PANDO_CONSUMER_ROOT}/src/geometry/native/src/sphericalpolygon.cpp")
foreach(domain document geometry map rendering data runtime)
  target_include_directories(pando-territorial-recipe PRIVATE "${PANDO_CONSUMER_ROOT}/src/${domain}/native/include")
endforeach()
target_link_libraries(pando-territorial-recipe PRIVATE PkgConfig::GEOS PkgConfig::PROJ)
if(MSVC)
  target_compile_options(pando-territorial-recipe PRIVATE /utf-8)
endif()
